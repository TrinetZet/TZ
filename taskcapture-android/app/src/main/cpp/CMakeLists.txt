cmake_minimum_required(VERSION 3.22.1)
project(taskcapture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Keep JNI debuggable while compiling model kernels with optimization.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
# Cross-compiling disables x86 instruction flags by default. CI emulator
# advertises SSE4.2, AVX and F16C; it does not advertise AVX2 or FMA.
if(ANDROID_ABI STREQUAL "x86_64")
 set(GGML_SSE42 ON CACHE BOOL "" FORCE)
 set(GGML_AVX ON CACHE BOOL "" FORCE)
 set(GGML_F16C ON CACHE BOOL "" FORCE)
 set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
 set(GGML_FMA OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama)
if(ANDROID)
 add_library(taskcapture SHARED engine.cpp taskcapture_jni.cpp)
 target_link_libraries(taskcapture PRIVATE llama android log)
else()
 add_executable(taskcapture-thread-test thread_config_test.cpp)
 add_executable(taskcapture-deadline-test deadline_test.cpp)
 add_executable(taskcapture-host engine.cpp host_main.cpp)
 target_link_libraries(taskcapture-host PRIVATE llama)
endif()
